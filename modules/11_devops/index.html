    Итоговый дипломный проект DevOps: Production-Ready Платформа «CloudOrder»

    Добро пожаловать в финальный дипломный проект курса! В этом модуле мы объединим знания и навыки из всех предыдущих 10 модулей (Linux, Bash, Git, Docker, SQL/PostgreSQL, Nginx, Observability, GitLab CI, CD/Rollback, Terraform, Ansible, Kubernetes и Helm) в единую сквозную инфраструктурную платформу уровня Production.

    Этот проект спроектирован так, чтобы вы могли: 1. Полностью развернуть и протестировать его локально на своей машине (через Docker, Terraform, Ansible, Kind и Helm). 2. Выложить готовый монорепозиторий в свой профиль на GitLab / GitHub как главное доказательство вашей квалификации на собеседованиях.

    1. Архитектура и сквозной релизный цикл (End-to-End Architecture)

    Мы создаем и автоматизируем отказоустойчивую платформу обработки заказов «CloudOrder API»:

    [ Разработчик: git push ]

    2. Исходный код сервиса (app/main.py):

    Сервис реализует: * Работу с PostgreSQL (хранение заказов) и Redis (кэширование счетчиков). * Поддержку Feature Flags (FEATURE_DISCOUNT_ENABLED). * Экспорт RED-метрик для Prometheus на эндпоинте /metrics (количество запросов по кодам ответа и гистограмма времени ответа). * Запись структурированных JSON-логов в stdout для парсинга в Grafana Loki.

    cat << 'EOF' > ~/devops-diploma/app/main.py
import json
import logging
import os
import socket
import time
import psycopg2
import redis
from flask import Flask, Response, jsonify, request
from prometheus_client import CONTENT_TYPE_LATEST, Counter, Histogram, generate_latest

app = Flask(__name__)

# Настройка структурированного JSON-логирования для Grafana Loki
logging.basicConfig(level=logging.INFO, format="%(message)s")
logger = logging.getLogger("cloudorder")

def log_json(level, event, **kwargs):
    payload = {
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "level": level,
        "service": "cloudorder-api",
        "env": os.getenv("ENV_NAME", "staging"),
        "version": os.getenv("APP_VERSION", "1.0.0"),
        "pod": socket.gethostname(),
        "event": event,
        **kwargs
    }
    logger.info(json.dumps(payload))

APP_VERSION = os.getenv("APP_VERSION", "1.0.0")
ENV_NAME = os.getenv("ENV_NAME", "staging")
FEATURE_DISCOUNT = os.getenv("FEATURE_DISCOUNT_ENABLED", "false").lower() == "true"
SIMULATE_ERROR = os.getenv("SIMULATE_ERROR", "false").lower() == "true"

# Метрики Prometheus (методология RED: Rate, Errors, Duration)
HTTP_REQUESTS_TOTAL = Counter(
    "cloudorder_http_requests_total",
    "Total count of HTTP requests",
    ["method", "endpoint", "status", "env", "version"]
)
HTTP_REQUEST_DURATION = Histogram(
    "cloudorder_http_request_duration_seconds",
    "HTTP request latency in seconds",
    ["endpoint", "env"]
)

def get_db_connection():
    return psycopg2.connect(
        host=os.getenv("DB_HOST", "localhost"),
        port=int(os.getenv("DB_PORT", 5432)),
        database=os.getenv("POSTGRES_DB", "orders_db"),
        user=os.getenv("POSTGRES_USER", "order_admin"),
        password=os.getenv("POSTGRES_PASSWORD", "StrongPass123"),
        connect_timeout=3
    )

def get_redis_client():
    return redis.Redis(
        host=os.getenv("REDIS_HOST", "localhost"),
        port=int(os.getenv("REDIS_PORT", 6379)),
        decode_responses=True
    )

@app.route("/health")
def health():
    if SIMULATE_ERROR:
        log_json("ERROR", "healthcheck_failed_simulated")
        HTTP_REQUESTS_TOTAL.labels("GET", "/health", "500", ENV_NAME, APP_VERSION).inc()
        return jsonify({"status": "unhealthy", "reason": "Simulated crash"}), 500
    try:
        conn = get_db_connection()
        conn.close()
        r = get_redis_client()
        r.ping()
        HTTP_REQUESTS_TOTAL.labels("GET", "/health", "200", ENV_NAME, APP_VERSION).inc()
        return jsonify({
            "status": "healthy",
            "version": APP_VERSION,
            "environment": ENV_NAME,
            "feature_discount": FEATURE_DISCOUNT,
            "pod": socket.gethostname()
        }), 200
    except Exception as exc:
        log_json("ERROR", "healthcheck_dependency_error", error=str(exc))
        HTTP_REQUESTS_TOTAL.labels("GET", "/health", "500", ENV_NAME, APP_VERSION).inc()
        return jsonify({"status": "unhealthy", "error": str(exc)}), 500

@app.route("/api/orders", methods=["GET", "POST"])
def handle_orders():
    start_time = time.time()
    if SIMULATE_ERROR:
        HTTP_REQUESTS_TOTAL.labels(request.method, "/api/orders", "500", ENV_NAME, APP_VERSION).inc()
        return jsonify({"error": "Simulated Internal Server Error"}), 500

    try:
        conn = get_db_connection()
        cur = conn.cursor()
        r = get_redis_client()

        cur.execute("""
            CREATE TABLE IF NOT EXISTS orders (
                id SERIAL PRIMARY KEY,
                item_name VARCHAR(120) NOT NULL,
                created_by_pod VARCHAR(64),
                app_version VARCHAR(32) DEFAULT '1.0.0',
                created_at TIMESTAMPTZ DEFAULT NOW()
            );
        """)

        if request.method == "POST":
            body = request.get_json(silent=True) or {}
            item_name = body.get("item", "DevOps-Pro-Subscription")
            cur.execute(
                "INSERT INTO orders (item_name, created_by_pod, app_version) VALUES (%s, %s, %s) RETURNING id;",
                (item_name, socket.gethostname(), APP_VERSION)
            )
            new_id = cur.fetchone()[0]
            conn.commit()
            r.incr("cloudorder_total_cached")
            log_json("INFO", "order_created", order_id=new_id, item=item_name)

        cur.execute("SELECT COUNT(*) FROM orders;")
        total_orders = cur.fetchone()[0]
        cur.close()
        conn.close()

        cached_orders = int(r.get("cloudorder_total_cached") or total_orders)
        duration = time.time() - start_time

        HTTP_REQUEST_DURATION.labels("/api/orders", ENV_NAME).observe(duration)
        HTTP_REQUESTS_TOTAL.labels(request.method, "/api/orders", "200", ENV_NAME, APP_VERSION).inc()

        resp = {
            "environment": ENV_NAME,
            "version": APP_VERSION,
            "served_by_pod": socket.gethostname(),
            "orders_in_postgres": total_orders,
            "orders_in_redis": cached_orders,
            "feature_discount_enabled": FEATURE_DISCOUNT
        }
        if FEATURE_DISCOUNT:
            resp["promo_banner"] = "VIP 20% Discount Active!"

        return jsonify(resp), 200
    except Exception as exc:
        log_json("ERROR", "order_query_failed", error=str(exc))
        HTTP_REQUESTS_TOTAL.labels(request.method, "/api/orders", "500", ENV_NAME, APP_VERSION).inc()
        return jsonify({"error": str(exc)}), 500

@app.route("/metrics")
def metrics():
    return Response(generate_latest(), mimetype=CONTENT_TYPE_LATEST)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=8000)
EOF

    Этап 2. SQL-миграции БД (migrations/V1__init_orders_and_dora.sql)

    Создадим идемпотентную миграцию по паттерну Expand & Contract: 

    cat << 'EOF' > ~/devops-diploma/migrations/V1__init_orders_and_dora.sql
BEGIN;

CREATE TABLE IF NOT EXISTS orders (
    id SERIAL PRIMARY KEY,
    item_name VARCHAR(120) NOT NULL,
    created_by_pod VARCHAR(64),
    app_version VARCHAR(32) DEFAULT '1.0.0',
    created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_orders_created_at ON orders (created_at DESC);

CREATE TABLE IF NOT EXISTS dora_deployments (
    id SERIAL PRIMARY KEY,
    environment VARCHAR(32) NOT NULL,
    version VARCHAR(32) NOT NULL,
    status VARCHAR(32) NOT NULL CHECK (status IN ('SUCCESS', 'ROLLED_BACK')),
    deployed_at TIMESTAMPTZ DEFAULT NOW()
);

COMMIT;
EOF

cat << 'EOF' > ~/devops-diploma/ansible/roles/node_hardening/defaults/main.yml

cat << 'EOF' > ~/devops-diploma/ansible/roles/node_hardening/templates/sysctl-k8s.conf.j2
# Managed by Ansible (CloudOrder Diploma)
# Node: {{ inventory_hostname }} | CPUs: {{ ansible_facts['processor_vcpus'] }} | RAM: {{ ansible_facts['memtotal_mb'] }} MB
net.ipv4.ip_forward = 1
fs.file-max = {{ max_open_files }}
vm.swappiness = 0
EOF

cat << 'EOF' > ~/devops-diploma/ansible/roles/node_hardening/tasks/main.yml
---
- name: Создать системный каталог конфигурации узла
  ansible.builtin.file:
    path: "{{ target_dir }}"
    state: directory
    mode: "0755"

- name: Сгенерировать параметры ядра для Kubernetes из шаблона Jinja2
  block:
    - name: Применить шаблон sysctl-k8s.conf.j2
      ansible.builtin.template:
        src: sysctl-k8s.conf.j2
        dest: "{{ target_dir }}/99-k8s-sysctl.conf"
        mode: "0644"
        backup: true
      register: sysctl_tpl

    - name: Проверить наличие обязательного параметра ip_forward = 1
